# Release Steps for BU BUDS v1.0.2

# 1. Make sure you're on the main branch and have the latest changes
git checkout main
git pull origin main

# 2. Stage all changes
git add .

# 3. Commit the changes with a descriptive message
git commit -m "Release version 1.0.2"

# 4. Create a new tag for the release
git tag -a v1.0.2 -m "Release version 1.0.2"

# 5. Push the tag to remote
git push origin v1.0.2

# 6. Push all changes to main branch
git push origin main
